function projectNote(note, notes) {
 if (!note) return null;
 return {
  ...note,
  incoming_links: Object.values(notes).filter(n => n.outgoing_links.some(link => link.slug === note.slug)),
  outgoing_links: note.outgoing_links.map(slug => notes[slug]).filter(Boolean)
 };
}

function filterNotes(notes, search, title, tag) {
 return Object.values(notes).filter(note => {
  const matchesSearch = !search || note.content.includes(search) || note.title.includes(search);
  const matchesTitle = !title || note.title.includes(title);
  const matchesTag = !tag || note.tags.includes(tag);
  return matchesSearch && matchesTitle && matchesTag;
 });
}
